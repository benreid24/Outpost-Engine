7/*
8 * General approach:
9 * 1. Generate a heightmap using Perlin noise
10 * 2. Generate a moisture map using Perlin noise
11 * 3. Use the heightmap and moisture map to determine the biome domain for each node
12 * 4. Perform domain collapse to select a biome for each node
13 * a. Start with the most constrained node
14 * b. Select a biome for the node based on the domain and weights
15 * c. Propagate the constraints to neighboring nodes?
16 * d. Repeat until all nodes have a selected biome
17 * 5. Postprocess
18 * a. Resample the heightmap for each node using parameters specific to the biome (can change
19 * octaves to get more/less jagged terrain, etc. Also option to scale height differently)
20 * b. Fill in lakes and rivers via BFS and raise nodes. TODO - how to handle downhill water?
21 * 6. Select sites for buildings and other structures
22 * a. Find or create flat areas
23 * b. Flatten sufficiently for the selected structure
24 * 7. Place train track via A*
25 * a. Select start and end points for the track
26 * b. Path through from start to end and visit all structures
27 * Note: Can handle bridges and tunnels by adding a large fixed cost based on slop to next node
28 * Cost would normally be distance and slope, but have a max threshold for slope
29 * 8. Place trees and rocks based on biome and random noise field
30 * Note: Poisson disk or jittered grid are an option, but can start with high freq Perlin
31 *
32 * TODO: Need a way to represent constraints. May want constraints to be additive in addition to
33 * restrictive sometimes
34 */

129void Generator::raiseLakes(ProtoTerrain& source) {
130 bl::ctr::Vector2D<std::uint8_t> visited(heightmap.getWidth(), heightmap.getHeight(), 0);
131
132 for (unsigned int x = 0; x < heightmap.getWidth(); ++x) {
133 for (unsigned int y = 0; y < heightmap.getHeight(); ++y) {
134 if (visited(x, y) != 0 || source.getNode(x, y).selectedBiome != Biome::Water) {
135 continue;
136 }
137
138 std::stack<glm::u32vec2> toVisit;
139 std::vector<glm::u32vec2> waterNodes;
140 float maxHeight = 0.f;
141
142 toVisit.push({x, y});
143 visited(x, y) = 1;
144
145 while (!toVisit.empty()) {
146 glm::u32vec2 current = toVisit.top();
147 toVisit.pop();
148
149 waterNodes.push_back(current);
150 maxHeight = std::max(maxHeight, heightmap(current.x, current.y));
151
152 for (const auto& neighbor : source.getNodeNeighbors(current.x, current.y)) {
153 if (visited(neighbor.position.x, neighbor.position.y) == 0 &&
154 neighbor.selectedBiome == Biome::Water) {
155 toVisit.push({neighbor.position.x, neighbor.position.y});
156 visited(neighbor.position.x, neighbor.position.y) = 1;
157 }
158 }
159 }
160
161 for (const auto& pos : waterNodes) {
162 heightmap(pos.x, pos.y) = maxHeight;
163 source.getNode(pos.x, pos.y).height = maxHeight;
164 toVisit.push(pos);
165 }
166
167 bl::ctr::Vector2D<std::uint8_t> visitedNonWater(
168 heightmap.getWidth(), heightmap.getHeight(), 0);
169 while (!toVisit.empty()) {
170 glm::u32vec2 current = toVisit.top();
171 toVisit.pop();
172
173 for (auto& neighbor : source.getNodeNeighbors(current.x, current.y)) {
174 if (visitedNonWater(neighbor.position.x, neighbor.position.y) == 0 &&
175 neighbor.height < maxHeight) {
176 visitedNonWater(neighbor.position.x, neighbor.position.y) = 1;
177 neighbor.selectedBiome = Biome::Water;
178 heightmap(neighbor.position.x, neighbor.position.y) = maxHeight;
179 neighbor.height = maxHeight;
