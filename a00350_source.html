35bl::ctr::StaticVector<Terrain::SampledBiome, 4> Terrain::sampleBiomes(const glm::vec2& pos) const {
36 const auto weightedPositions = getPositionWeights(pos);
37 bl::ctr::StaticVector<SampledBiome, 4> result;
38 for (const auto& wp : weightedPositions) {
39 const Node& node = nodes(wp.first.x, wp.first.y);
40 for (auto& sample : result) {
41 if (sample.biome == node.biome) {
42 const float tw = sample.weight + wp.second;
43 sample.height = sample.height * sample.weight / tw + node.height * wp.second / tw;
44 sample.weight += wp.second;
45 goto next;
46 }
47 }
48 result.emplace_back(SampledBiome{node.biome, node.height, wp.second});
49 next:;
50 }
51 return result;
52}

54std::array<std::pair<glm::u32vec2, float>, 4> Terrain::getPositionWeights(
55 const glm::vec2& pos) const {
56 const glm::vec2 normalPos = pos / worldSize + glm::vec2(0.5f);
57 if (normalPos.x < 0.f || normalPos.x > 1.f || normalPos.y < 0.f || normalPos.y > 1.f) {
58 return {};
59 }
60 const float xi = normalPos.x * static_cast<float>(nodes.getWidth() - 1);
61 const float yi = normalPos.y * static_cast<float>(nodes.getHeight() - 1);
62 const unsigned int x0 = static_cast<unsigned int>(std::floor(xi));
63 const unsigned int y0 = static_cast<unsigned int>(std::floor(yi));
64 const unsigned int x1 = std::min(x0 + 1, nodes.getWidth() - 1);
65 const unsigned int y1 = std::min(y0 + 1, nodes.getHeight() - 1);
66 const float xn = xi - static_cast<float>(x0);
67 const float yn = yi - static_cast<float>(y0);
68 const float w00 = (1.f - xn) * (1.f - yn);
69 const float w10 = xn * (1.f - yn);
70 const float w01 = (1.f - xn) * yn;
71 const float w11 = xn * yn;
72 return {{{{x0, y0}, w00}, {{x1, y0}, w10}, {{x0, y1}, w01}, {{x1, y1}, w11}}};
73}

101 for (unsigned int i = 0; i < chunks; ++i) {
102 futures.emplace_back(threadPool.queueTask([this, &verts, i, chunkSize] {
