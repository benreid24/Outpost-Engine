54void ArenaController::update(float dt) {
55 debounce -= dt;
56
57 // adjust scalars
58 updateValue(currentDistance, targetDistance, dt);
59 updateValue(currentPitch, targetPitch, dt);
60
61 // update yaw via shortest path
62 const float yawDiff = targetYaw - currentYaw;
63 if (yawDiff > 180.f) { currentYaw += 360.f; }
64 else if (yawDiff < -180.f) { currentYaw -= 360.f; }
65 updateValue(currentYaw, targetYaw, dt);
66
67 // adjust velocity for yaw so we move in camera space
68 const float yawCos = std::cos(glm::radians(currentYaw));
69 const float yawSin = std::sin(glm::radians(currentYaw));
70 const float pitchCos = std::cos(glm::radians(currentPitch));
71 const float pitchSin = std::sin(glm::radians(currentPitch));
72
73 // update focus position
74 const glm::vec3 yawAdjustedVelocity(velocity.x * yawCos + velocity.z * yawSin,
75 velocity.y,
76 velocity.z * yawCos - velocity.x * yawSin);
77 speed = speedPerStep * static_cast<float>(currentDistanceStep + 1);
78 currentPosition += yawAdjustedVelocity * speed * speedMultiple * dt;
79 currentPosition.y = std::max(currentPosition.y, 0.f);
80
81 // update camera position and orientation
82 glm::vec3 camPos;
83 camPos.x = currentPosition.x + yawSin * pitchCos * currentDistance;
84 camPos.z = currentPosition.z + yawCos * pitchCos * currentDistance;
85 const float terrainHeight = arena.getTerrain().sampleHeight({camPos.x, camPos.z});
86 camPos.y = currentPosition.y + pitchSin * currentDistance;
87 camPos.y = std::max(camPos.y, terrainHeight + 1.f);
88 camera().setPosition(camPos);
89 camera().getOrientationForChange().lookAt(currentPosition, camPos);
90
91 // update velocity
92 updateVelocity(velocity.x, targetVelocity.x, dampening, dt);
93 updateVelocity(velocity.y, targetVelocity.y, dampening, dt);
94 updateVelocity(velocity.z, targetVelocity.z, dampening, dt);
95}
