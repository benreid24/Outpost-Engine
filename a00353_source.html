37void Car::update(const Car& connectedTo, const Track& track) {
38 // distance from rear axle to rear of car for the one we are connected to
39 const float connectedCouplerOffset =
40 connectedTo.config.length * 0.5f - connectedTo.config.rearAxleOffset;
41
42 // distance from front axle to front of car for this one
43 const float ourCouplerOffset = config.length * 0.5f - config.frontAxleOffset;

60void Car::updateSpatialState(const Track& track) {
61 const glm::vec3 frontAxlePos = track.getPositionAtDistance(state.frontAxleTrackPosition);
62 const glm::vec3 frontAxleDir = track.getRightAtDistance(state.frontAxleTrackPosition);
63 const glm::vec3 frontAxleUp = track.getUpAtDistance(state.frontAxleTrackPosition);
64 state.frontWheelWorldTransform.setPosition(frontAxlePos +
65 frontAxleUp * config.wheelHeight * 0.5f);
66 state.frontWheelWorldTransform.setForwardDir(frontAxleDir, frontAxleUp);
67
68 const glm::vec3 rearAxlePos = track.getPositionAtDistance(state.rearAxleTrackPosition);
69 const glm::vec3 rearAxleDir = track.getRightAtDistance(state.rearAxleTrackPosition);
70 const glm::vec3 rearAxleUp = track.getUpAtDistance(state.rearAxleTrackPosition);
71 state.rearWheelWorldTransform.setForwardDir(rearAxleDir, rearAxleUp);
72 state.rearWheelWorldTransform.setPosition(rearAxlePos + rearAxleUp * config.wheelHeight * 0.5f);
73
74 const glm::vec3 rearWheelAttachPos =
75 rearAxlePos + rearAxleUp * config.wheelHeight * WheelEmbedFactor;
76 const glm::vec3 frontWheelAttachPos =
77 frontAxlePos + frontAxleUp * config.wheelHeight * WheelEmbedFactor;
78
79 const glm::vec3 rearToFront = glm::normalize(frontWheelAttachPos - rearWheelAttachPos);
80 const glm::vec3 carUp = glm::normalize((frontAxleUp + rearAxleUp) * 0.5f);
81 const float carHeight = config.wheelHeight * WheelEmbedFactor + config.height * 0.5f;
82 glm::vec3 carWorldPos = rearAxlePos + rearToFront * config.rearAxleOffset;
83 carWorldPos += carUp * carHeight;
84 state.carWorldTransform.setPosition(carWorldPos);
85 state.carWorldTransform.setForwardDir(frontAxlePos - rearAxlePos, carUp);
86
87 if (carBody.exists()) {
88 carBody.getTransform() = state.carWorldTransform;
89 frontWheel.getTransform() = state.frontWheelWorldTransform;
